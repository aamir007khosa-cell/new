/* Tower Finder — license backend (Cloudflare Worker)
   Verifies a USDT payment on-chain, then signs a device-bound key and
   publishes it to keys.json so the customer's app auto-activates.

   Secrets (Worker → Settings → Variables & Secrets):
     PRV_JWK  = the ECDSA P-256 private key as a JSON string (keys/prv.json)
     GH_TOKEN = GitHub fine-grained token with Contents:Read+Write on repo "new"
     ADMIN_KEY = a password you choose; the admin page uses it to revoke/unrevoke devices
   No KV needed: spent TXIDs are tracked inside keys.json ("used" array).
*/

const CFG = {
  pricePerDayUsdt: 3.5,
  maxDays: 30,
  amountTolerance: 0.05,     // allow tiny rounding under the expected amount
  maxAgeSec: 7 * 86400,      // payment must be at most this old
  gh: { owner: 'aamir007khosa-cell', repo: 'new', path: 'keys.json' },
  trc20: {
    wallet: 'TARezKnmnzeaWewvap9EEpuhnZMMHVbbru',
    contract: 'TR7NHqjeKQxGTCi8q8ZY4pL8otSzgjLj6t',   // USDT on TRON
    walletHex: '05008215de732bfb0abd164e642c5b40bb0f7da8',
    contractHex: 'a614f803b6fd780986a42c78ec9c7f77e6ded13c',
    decimals: 6,
  },
  bep20: {
    wallet: '0xF2F9C8586E2D9180e75B825Bed53f1F2fF6E31E8',
    contract: '0x55d398326f99059fF775485246999027B3197955', // USDT on BSC
    decimals: 18,
    rpcs: ['https://bsc-dataseed.binance.org', 'https://bsc-dataseed1.ninicoin.io', 'https://bsc.publicnode.com'],
  },
};
const TRANSFER_TOPIC = '0xddf252ad1be2c89b69c2b068fc378daa952ba7f163c4a11628f55a4df523b3ef';

const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type',
};
const json = (obj, status = 200) =>
  new Response(JSON.stringify(obj), { status, headers: { 'Content-Type': 'application/json', ...CORS } });

export default {
  async fetch(req, env) {
    if (req.method === 'OPTIONS') return new Response(null, { headers: CORS });
    const url = new URL(req.url);
    if (req.method !== 'POST') return json({ ok: false, error: 'Use POST.' }, 405);

    let body = {};
    try { body = await req.json(); } catch (_) { return json({ ok: false, error: 'Bad JSON.' }, 400); }

    const path = url.pathname.replace(/\/+$/, '');

    // ----- admin: password check -----
    if (path.endsWith('/ping')) {
      if (!env.ADMIN_KEY || String(body.admin || '') !== String(env.ADMIN_KEY)) return json({ ok: false, error: 'Wrong admin password.' }, 401);
      return json({ ok: true });
    }
    // ----- admin: revoke / unrevoke a device (instant remote logout) -----
    if (path.endsWith('/revoke') || path.endsWith('/unrevoke')) {
      return handleRevoke(env, body, path.endsWith('/unrevoke'));
    }
    // ----- Easypaisa / JazzCash manual orders -----
    if (path.endsWith('/order')) return handleOrder(env, body);
    if (path.endsWith('/orders') || path.endsWith('/approve') || path.endsWith('/reject') || path.endsWith('/shot')) {
      return handleOrderAdmin(env, body, path.split('/').pop());
    }
    // ----- admin: publish / unpublish a locally-signed key (no payment) -----
    if (path.endsWith('/publish') || path.endsWith('/unpublish')) {
      return handlePublish(env, body, path.endsWith('/unpublish'));
    }

    const dev = String(body.dev || '').trim().toUpperCase();
    const txid = String(body.txid || '').trim();
    const net = String(body.net || '').trim().toLowerCase();   // 'trc20' | 'bep20'
    const days = parseInt(body.days, 10);
    const debug = url.pathname.replace(/\/+$/, '').endsWith('/check');

    if (!/^[A-Z0-9]{8}$/.test(dev)) return json({ ok: false, error: 'Invalid device code.' });
    if (!(days >= 1 && days <= CFG.maxDays)) return json({ ok: false, error: 'Invalid package.' });
    if (net !== 'trc20' && net !== 'bep20') return json({ ok: false, error: 'Invalid network.' });
    if (txid.length < 20) return json({ ok: false, error: 'Invalid TXID.' });

    const required = Math.round(days * CFG.pricePerDayUsdt * 100) / 100;
    const txKey = net + ':' + txid.toLowerCase();

    // 1) fast pre-check: already-used TXID
    if (!debug) {
      try {
        const f = await getKeysFile(env.GH_TOKEN);
        if ((f.data.used || []).includes(txKey)) return json({ ok: false, error: 'This Transaction ID was already used.' });
      } catch (_) { /* non-fatal; the atomic check below still guards */ }
    }

    // 2) verify on-chain
    let v;
    try {
      v = net === 'trc20' ? await verifyTron(txid) : await verifyBsc(txid);
    } catch (e) {
      return json({ ok: false, error: 'Could not read the transaction. Try again in a minute.', detail: String(e).slice(0, 140) });
    }
    if (!v.found) return json({ ok: false, error: 'Transaction not found yet. Wait for confirmation and retry.' });
    if (!v.confirmed) return json({ ok: false, error: 'Payment not confirmed yet. Wait a minute and retry.' });
    if (!v.toMatch) return json({ ok: false, error: 'Payment did not go to our wallet.' });
    if (v.ageSec > CFG.maxAgeSec) return json({ ok: false, error: 'This payment is too old.' });
    if (v.amount + CFG.amountTolerance < required)
      return json({ ok: false, error: 'Amount too low. Paid ' + v.amount + ' USDT, need ' + required + ' USDT.' });

    if (debug) return json({ ok: true, debug: true, verified: v, required });

    // 3) sign device-bound key
    let key;
    try { key = await signKey(env.PRV_JWK, dev, days); }
    catch (e) { return json({ ok: false, error: 'Signing failed.', detail: String(e).slice(0, 140) }); }

    // 4) commit atomically: re-check used, record key + spent TXID in keys.json
    try {
      const r = await commitKeys(env.GH_TOKEN, (data) => {
        data.used = data.used || [];
        if (data.used.includes(txKey)) return { abort: true };
        data.list = (data.list || []).filter(e => e.d !== dev);
        data.list.push({ d: dev, k: key, t: Math.floor(Date.now() / 1000) });
        data.used.push(txKey);
        if (data.list.length > 800) data.list = data.list.slice(-800);
        if (data.used.length > 4000) data.used = data.used.slice(-4000);
      });
      if (r && r.abort) return json({ ok: false, error: 'This Transaction ID was already used.' });
    } catch (e) {
      return json({ ok: false, error: 'Could not save the key. Try again in a minute.', detail: String(e).slice(0, 140) });
    }

    return json({ ok: true, key, expiresInDays: days });
  },
};

/* ---------- TRON (Tronscan) ---------- */
async function tronFetch(txid) {
  const hosts = ['https://apilist.tronscanapi.com', 'https://api.tronscan.org'];
  let lastErr = 'no data';
  for (let attempt = 0; attempt < 3; attempt++) {
    for (const h of hosts) {
      try {
        const r = await fetch(h + '/api/transaction-info?hash=' + encodeURIComponent(txid), { headers: { Accept: 'application/json' } });
        if (r.status === 429) { lastErr = '429'; continue; }
        if (!r.ok) { lastErr = String(r.status); continue; }
        const d = await r.json();
        if (d && (d.hash || d.contractData)) return d;
        lastErr = 'empty';
      } catch (e) { lastErr = String(e).slice(0, 40); }
    }
    if (attempt < 2) await new Promise(res => setTimeout(res, 500));
  }
  throw new Error('tronscan ' + lastErr);
}
async function verifyTron(txid) {
  const out = { found: false, confirmed: false, toMatch: false, amount: 0, ageSec: 1e9 };
  // Primary: TronGrid (EVM-style logs, reliable from Workers)
  try {
    const r = await fetch('https://api.trongrid.io/wallet/gettransactioninfobyid', {
      method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ value: txid }),
    });
    if (r.ok) {
      const j = await r.json();
      if (j && j.id) {
        out.found = true;
        out.confirmed = j.receipt && j.receipt.result ? j.receipt.result === 'SUCCESS' : true;
        if (j.blockTimeStamp) out.ageSec = Math.max(0, Math.floor(Date.now() / 1000 - j.blockTimeStamp / 1000));
        const topic = TRANSFER_TOPIC.replace(/^0x/, '');
        for (const log of (j.log || [])) {
          if ((log.address || '').toLowerCase() !== CFG.trc20.contractHex) continue;
          if (!log.topics || (log.topics[0] || '').toLowerCase().replace(/^0x/, '') !== topic) continue;
          if ((log.topics[2] || '').toLowerCase().slice(-40) !== CFG.trc20.walletHex) continue;
          const amt = Number(BigInt('0x' + log.data)) / Math.pow(10, CFG.trc20.decimals);
          if (amt > out.amount) { out.amount = amt; out.toMatch = true; }
        }
        return out;
      }
    }
  } catch (_) { /* fall through to Tronscan */ }
  // Fallback: Tronscan
  const j = await tronFetch(txid);
  if (!j || (!j.hash && !j.contractData)) return out;
  out.found = true;
  out.confirmed = j.confirmed === true && (j.contractRet === 'SUCCESS' || j.result === 'SUCCESS' || !j.contractRet);
  if (j.timestamp) out.ageSec = Math.max(0, Math.floor(Date.now() / 1000 - j.timestamp / 1000));
  // transfers can be in trc20TransferInfo (array) or tokenTransferInfo (object)
  let xfers = [];
  if (Array.isArray(j.trc20TransferInfo)) xfers = j.trc20TransferInfo;
  else if (j.tokenTransferInfo) xfers = [j.tokenTransferInfo];
  for (const t of xfers) {
    const to = t.to_address || t.toAddress;
    const contract = t.contract_address || t.contractAddress;
    if (!to || !contract) continue;
    if (contract !== CFG.trc20.contract) continue;
    if (to !== CFG.trc20.wallet) continue;
    const dec = +(t.decimals != null ? t.decimals : CFG.trc20.decimals);
    const raw = t.amount_str || t.amount || t.quant || '0';
    const amt = Number(raw) / Math.pow(10, dec);
    if (amt > out.amount) { out.amount = amt; out.toMatch = true; }
  }
  return out;
}

/* ---------- BSC (public RPC) ---------- */
async function rpc(method, params) {
  let lastErr;
  for (const u of CFG.bep20.rpcs) {
    try {
      const r = await fetch(u, {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ jsonrpc: '2.0', id: 1, method, params }),
      });
      if (!r.ok) { lastErr = new Error(method + ' ' + r.status); continue; }
      const j = await r.json();
      if (j.error) { lastErr = new Error(j.error.message || 'rpc error'); continue; }
      return j.result;
    } catch (e) { lastErr = e; }
  }
  throw lastErr || new Error('rpc failed');
}
function padAddr(a) { return '0x' + '0'.repeat(24) + a.toLowerCase().replace(/^0x/, ''); }
async function verifyBsc(txid) {
  const out = { found: false, confirmed: false, toMatch: false, amount: 0, ageSec: 1e9 };
  const rcpt = await rpc('eth_getTransactionReceipt', [txid]);
  if (!rcpt) return out;
  out.found = true;
  out.confirmed = rcpt.status === '0x1';
  const wallet = padAddr(CFG.bep20.wallet);
  const contract = CFG.bep20.contract.toLowerCase();
  for (const log of (rcpt.logs || [])) {
    if ((log.address || '').toLowerCase() !== contract) continue;
    if (!log.topics || log.topics[0] !== TRANSFER_TOPIC) continue;
    if ((log.topics[2] || '').toLowerCase() !== wallet) continue;
    const amt = Number(BigInt(log.data)) / Math.pow(10, CFG.bep20.decimals);
    if (amt > out.amount) { out.amount = amt; out.toMatch = true; }
  }
  // age via block timestamp
  if (rcpt.blockNumber) {
    try {
      const blk = await rpc('eth_getBlockByNumber', [rcpt.blockNumber, false]);
      if (blk && blk.timestamp) out.ageSec = Math.max(0, Math.floor(Date.now() / 1000 - Number(BigInt(blk.timestamp))));
    } catch (_) { out.ageSec = 0; }
  }
  return out;
}

/* ---------- signing ---------- */
function b64uEnc(buf) {
  let s = ''; const b = new Uint8Array(buf);
  for (let i = 0; i < b.length; i++) s += String.fromCharCode(b[i]);
  return btoa(s).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}
async function signKey(prvJwkStr, dev, days) {
  const jwk = JSON.parse(prvJwkStr);
  const sk = await crypto.subtle.importKey('jwk', jwk, { name: 'ECDSA', namedCurve: 'P-256' }, false, ['sign']);
  const e = Math.floor(Date.now() / 1000) + days * 86400;
  const pStr = b64uEnc(new TextEncoder().encode(JSON.stringify({ d: dev, e, p: days })));
  const sig = await crypto.subtle.sign({ name: 'ECDSA', hash: 'SHA-256' }, sk, new TextEncoder().encode(pStr));
  return pStr + '.' + b64uEnc(sig);
}

/* ---------- publish to keys.json ---------- */
async function gh(token, method, path, bodyObj) {
  const r = await fetch('https://api.github.com/repos/' + CFG.gh.owner + '/' + CFG.gh.repo + '/contents/' + path + (method === 'GET' ? '?t=' + Date.now() : ''), {
    method,
    headers: { Authorization: 'Bearer ' + token, Accept: 'application/vnd.github+json', 'User-Agent': 'tf-worker', ...(bodyObj ? { 'Content-Type': 'application/json' } : {}) },
    body: bodyObj ? JSON.stringify(bodyObj) : undefined,
  });
  return r;
}
async function getFile(token, path, dflt) {
  const g = await gh(token, 'GET', path);
  if (g.status === 404) return { data: JSON.parse(JSON.stringify(dflt)), sha: null };
  if (g.status !== 200) throw new Error('gh read ' + g.status);
  const j = await g.json();
  return { data: JSON.parse(decodeURIComponent(escape(atob(j.content.replace(/\n/g, ''))))), sha: j.sha };
}
// read-modify-write with one retry on a concurrent-edit (409) conflict
async function commitFile(token, path, dflt, mutate, msg) {
  for (let attempt = 0; attempt < 2; attempt++) {
    const { data, sha } = await getFile(token, path, dflt);
    const res = await mutate(data);
    if (res && res.abort) return res;
    const content = btoa(unescape(encodeURIComponent(JSON.stringify(data) + '\n')));
    const put = await gh(token, 'PUT', path, { message: msg || 'update', content, ...(sha ? { sha } : {}) });
    if (put.ok) return { ok: true };
    if (put.status === 409) continue;
    throw new Error('gh write ' + put.status);
  }
  throw new Error('gh write conflict');
}
async function getKeysFile(token) { return getFile(token, CFG.gh.path, { list: [], used: [] }); }
async function commitKeys(token, mutate) { return commitFile(token, CFG.gh.path, { list: [], used: [] }, mutate, 'Auto key'); }

/* ---------- admin: remote logout via revoked.json ---------- */
async function handleRevoke(env, body, isUn) {
  if (!env.ADMIN_KEY) return json({ ok: false, error: 'Admin not configured (set ADMIN_KEY).' }, 500);
  if (String(body.admin || '') !== String(env.ADMIN_KEY)) return json({ ok: false, error: 'Wrong admin password.' }, 401);
  const dev = String(body.dev || '').trim().toUpperCase();
  if (!/^[A-Z0-9]{8}$/.test(dev)) return json({ ok: false, error: 'Invalid device code.' });
  const now = Math.floor(Date.now() / 1000);
  try {
    await commitFile(env.GH_TOKEN, 'revoked.json', { v: 1, list: [] }, (data) => {
      data.v = 1;
      data.list = (data.list || []).filter(e => e && e.d !== dev);
      if (!isUn) data.list.push({ d: dev, t: now });
      if (data.list.length > 2000) data.list = data.list.slice(-2000);
    }, (isUn ? 'Unrevoke ' : 'Revoke ') + dev);
  } catch (e) {
    return json({ ok: false, error: 'Could not update. Try again.', detail: String(e).slice(0, 140) });
  }
  return json({ ok: true, dev, revoked: !isUn });
}

/* ---------- admin: publish a locally-signed key to keys.json (local/physical deals) ---------- */
async function handlePublish(env, body, isUn) {
  if (!env.ADMIN_KEY) return json({ ok: false, error: 'Admin not configured (set ADMIN_KEY).' }, 500);
  if (String(body.admin || '') !== String(env.ADMIN_KEY)) return json({ ok: false, error: 'Wrong admin password.' }, 401);
  const dev = String(body.dev || '').trim().toUpperCase();
  if (!/^[A-Z0-9]{8}$/.test(dev)) return json({ ok: false, error: 'Invalid device code.' });
  const key = String(body.key || '').trim();
  if (!isUn && key.length < 20) return json({ ok: false, error: 'Missing key.' });
  try {
    await commitKeys(env.GH_TOKEN, (data) => {
      data.list = (data.list || []).filter(e => e.d !== dev);
      if (!isUn) {
        data.list.push({ d: dev, k: key, t: Math.floor(Date.now() / 1000) });
        if (data.list.length > 800) data.list = data.list.slice(-800);
      }
    });
  } catch (e) {
    return json({ ok: false, error: 'Could not save. Try again.', detail: String(e).slice(0, 140) });
  }
  return json({ ok: true, dev });
}

/* ---------- Easypaisa / JazzCash orders (encrypted at rest in orders.enc.json) ---------- */
const ORDERS_PATH = 'orders.enc.json';
async function ordKey(env) {
  const h = await crypto.subtle.digest('SHA-256', new TextEncoder().encode('orders:' + env.ADMIN_KEY));
  return crypto.subtle.importKey('raw', h, 'AES-GCM', false, ['encrypt', 'decrypt']);
}
const b64 = (u8) => { let r = ''; for (let i = 0; i < u8.length; i += 0x4000) r += String.fromCharCode.apply(null, u8.subarray(i, i + 0x4000)); return btoa(r); };
const unb64 = (s) => Uint8Array.from(atob(s), c => c.charCodeAt(0));
async function ordDecode(env, data) {
  if (!data || !data.ct) return [];
  try {
    const pt = await crypto.subtle.decrypt({ name: 'AES-GCM', iv: unb64(data.iv) }, await ordKey(env), unb64(data.ct));
    return JSON.parse(new TextDecoder().decode(pt));
  } catch (_) { return []; }
}
async function ordEncode(env, list) {
  const iv = crypto.getRandomValues(new Uint8Array(12));
  const ct = new Uint8Array(await crypto.subtle.encrypt({ name: 'AES-GCM', iv }, await ordKey(env), new TextEncoder().encode(JSON.stringify(list))));
  return { v: 1, iv: b64(iv), ct: b64(ct) };
}
async function mutateOrders(env, fn, msg) {
  let out;
  const r = await commitFile(env.GH_TOKEN, ORDERS_PATH, {}, async (data) => {
    const list = await ordDecode(env, data);
    const res = fn(list);
    if (res && res.abort) return res;
    out = res;
    const enc = await ordEncode(env, list);
    data.v = enc.v; data.iv = enc.iv; data.ct = enc.ct;
  }, msg || 'order');
  return r && r.abort ? r : { ok: true, out };
}

async function handleOrder(env, body) {
  if (!env.ADMIN_KEY) return json({ ok: false, error: 'Not configured.' }, 500);
  const dev = String(body.dev || '').trim().toUpperCase();
  const days = parseInt(body.days, 10);
  const method = String(body.method || '').toLowerCase();
  let txid = String(body.txid || '').trim().replace(/\s+/g, '').slice(0, 40);
  const shot = String(body.shot || '');
  const name = String(body.name || '').trim().slice(0, 40);
  const phone = String(body.phone || '').trim().slice(0, 20);
  if (!/^[A-Z0-9]{8}$/.test(dev)) return json({ ok: false, error: 'Invalid device code.' });
  if (!(days >= 1 && days <= CFG.maxDays)) return json({ ok: false, error: 'Invalid package.' });
  if (method !== 'easypaisa' && method !== 'jazzcash') return json({ ok: false, error: 'Invalid method.' });
  if (shot) {
    if (!/^data:image\/jpeg;base64,[A-Za-z0-9+\/=]+$/.test(shot) || shot.length > 450000) return json({ ok: false, error: 'Screenshot is invalid or too large.' });
    const h = new Uint8Array(await crypto.subtle.digest('SHA-256', new TextEncoder().encode(shot)));
    txid = 'S' + Array.from(h).slice(0, 8).map(x => x.toString(16).padStart(2, '0')).join('');
  } else if (!/^[A-Za-z0-9]{6,40}$/.test(txid)) return json({ ok: false, error: 'Please upload your payment screenshot.' });
  const id = 'O' + Date.now().toString(36).toUpperCase() + Math.random().toString(36).slice(2, 5).toUpperCase();
  try {
    if (shot) {
      await commitFile(env.GH_TOKEN, 'shots/' + id + '.enc.json', {}, async (data) => {
        const enc = await ordEncode(env, shot);
        data.v = enc.v; data.iv = enc.iv; data.ct = enc.ct;
      }, 'shot');
    }
    const r = await mutateOrders(env, (list) => {
      if (list.some(o => o.txid.toLowerCase() === txid.toLowerCase())) return { abort: true };
      list.push({ id, dev, days, method, txid, name, phone, shot: !!shot, t: Math.floor(Date.now() / 1000), status: 'pending' });
      while (list.length > 400) list.shift();
    }, 'order');
    if (r.abort) return json({ ok: false, error: shot ? 'This screenshot was already submitted.' : 'This Transaction ID was already submitted.' });
  } catch (e) {
    return json({ ok: false, error: 'Could not save. Try again in a minute.', detail: String(e).slice(0, 100) });
  }
  return json({ ok: true, id });
}

async function handleOrderAdmin(env, body, action) {
  if (!env.ADMIN_KEY) return json({ ok: false, error: 'Admin not configured.' }, 500);
  if (String(body.admin || '') !== String(env.ADMIN_KEY)) return json({ ok: false, error: 'Wrong admin password.' }, 401);
  try {
    if (action === 'shot') {
      const sid = String(body.id || '').replace(/[^A-Z0-9]/gi, '');
      const f = await getFile(env.GH_TOKEN, 'shots/' + sid + '.enc.json', {});
      const d = await ordDecode(env, f.data);
      return json(typeof d === 'string' && d ? { ok: true, shot: d } : { ok: false, error: 'No screenshot found.' });
    }
    if (action === 'orders') {
      const f = await getFile(env.GH_TOKEN, ORDERS_PATH, {});
      return json({ ok: true, orders: (await ordDecode(env, f.data)).slice().reverse() });
    }
    const id = String(body.id || '');
    if (action === 'reject') {
      const r = await mutateOrders(env, (list) => {
        const o = list.find(x => x.id === id); if (!o) return { abort: true };
        o.status = 'rejected';
      }, 'reject');
      return json(r.abort ? { ok: false, error: 'Order not found.' } : { ok: true });
    }
    // approve: sign key, publish, mark approved
    const f = await getFile(env.GH_TOKEN, ORDERS_PATH, {});
    const o = (await ordDecode(env, f.data)).find(x => x.id === id);
    if (!o) return json({ ok: false, error: 'Order not found.' });
    if (o.status === 'approved') return json({ ok: true, already: true });
    const key = await signKey(env.PRV_JWK, o.dev, o.days);
    await commitKeys(env.GH_TOKEN, (data) => {
      data.list = (data.list || []).filter(e => e.d !== o.dev);
      data.list.push({ d: o.dev, k: key, t: Math.floor(Date.now() / 1000) });
      if (data.list.length > 800) data.list = data.list.slice(-800);
    });
    await mutateOrders(env, (list) => { const x = list.find(z => z.id === id); if (x) x.status = 'approved'; }, 'approve');
    return json({ ok: true, dev: o.dev, days: o.days });
  } catch (e) {
    return json({ ok: false, error: 'Failed. Try again.', detail: String(e).slice(0, 100) });
  }
}
